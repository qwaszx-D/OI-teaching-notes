\chapter{决策单调性}

\section{基本概念和应用}

最常见的情况是一维动态规划的决策单调性。
\begin{definition}[一维决策单调性]
考虑 $f_i=\min_{j<i} A(j,i)$ 形式的一维动态规划。定义 $i$ 的决策点为
$$
\mathrm{opt}(i)=\min\arg\min_{j<i}A(j,i),
$$
即取到最值的 $j$ 中最小的一个。若 $\mathrm{opt}(i)$ 单调不降，则称该动态规划具有\textbf{决策单调性}。
\end{definition}

对于 $A$ 不依赖 $f$ 的情况，可以使用分治来快速计算 $f$。

\begin{descbox}{算法}[决策单调性分治]
    $\mathtt{solve}(l,r,L,R)$ 表示当前需要计算 $[l,r]$ 内的 $f_i$，且已知决策点在 $[L,R]$ 内。令 $m=\lfloor(l+r)/2\rfloor$，$O(R-L+1)$ 计算 $f_m$ 的最优决策点 $p$，递归处理 $\mathtt{solve}(l,m-1,L,p)$ 和 $\mathtt{solve}(m+1,r,p,R)$ 即可。
    
    考虑分治树的每一层。递归到下一层时，$\sum R-L+1$ 的增加量为该层的结点个数。初始的 $R-L+1=n$，这部分总共贡献 $O(n\log n)$；第 $i$ 层增加的 $2^i$ 个结点会贡献 $\log_2 n-i+1$ 次，这部分总贡献 $O(n)$。于是总复杂度 $O(n\log n)$。
\end{descbox}
\begin{exercise}
    考虑算法过程中依次询问的所有 $A(j_k,i_k)$。证明 $\sum_k |j_k-j_{k-1}|+|i_k-i_{k-1}|=O(n\log n)$。
\end{exercise}
\begin{exercise}
    给定一个排列，将其划分为 $k$ 段，每段的代价为该段的逆序对数。最小化总代价。在已知此问题具有决策单调性的情况下，设计 $O(nK\log^2 n)$ 的算法。
\end{exercise}

\section{Monge 性质}

我们考虑一种被称为 Monge 性质的凸性，
\begin{definition}[Monge 矩阵]
    对于矩阵 $A$，若其混合差分
    $$
    \Delta_{xy}A(i,j)=A(i+1,j+1)-A(i+1,j)-A(i,j+1)+A(i,j)
    $$
    非正，则称 $A$ 为 Monge 矩阵。
\end{definition}

\begin{lemma}
    以下命题等价：
    \begin{itemize}
        \item $A$ 是 Monge 矩阵
        \item $A$ 的行差分 $\Delta_x A(i,j)=A(i+1,j)-A(i,j)$ 关于列 $j$ 不增
        \item $A$ 的列差分 $\Delta_y A(i,j)=A(i,j+1)-A(i,j)$ 关于行 $i$ 不增
        \item 四边形不等式：对任意 $i_1\le i_2,j_1\le j_2$，都有 $A(i_1,j_1)+A(i_2,j_2)\le A(i_1,j_2)+A(i_2,j_1)$
    \end{itemize}
\end{lemma}
